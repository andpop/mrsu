    SELECT
        r.user_id,
        r.movie_id,
        r.rating,
        r.timestamp,
        ROW_NUMBER() OVER (
            PARTITION BY r.user_id
            ORDER BY r.timestamp DESC
        ) AS rn
    FROM ratings r
)
SELECT
    m.title,
    u.name AS user_name,
    lr.rating,
    date(lr.timestamp, 'unixepoch') AS rating_date
FROM latest_ratings lr
JOIN users u ON u.id = lr.user_id
JOIN movies m ON m.id = lr.movie_id
WHERE lr.rn = 1
ORDER BY lr.timestamp DESC
LIMIT 10;"
echo " "

sqlite3 movies_rating.db -box -echo "WITH movie_averages AS (
    SELECT
        m.id,
        m.title,
        m.year,
        AVG(r.rating) AS avg_rating
    FROM movies m
    JOIN ratings r ON r.movie_id = m.id
    GROUP BY m.id, m.title, m.year
),
rating_limits AS (
    SELECT
        MAX(avg_rating) AS max_rating,
        MIN(avg_rating) AS min_rating
    FROM movie_averages
)
SELECT
    ma.title,
    ma.year,
    ROUND(ma.avg_rating, 3) AS avg_rating,
    CASE
        WHEN ma.avg_rating = rl.max_rating THEN 'Да'
        ELSE 'Нет'
    END AS 'Рекомендуем'
FROM movie_averages ma
CROSS JOIN rating_limits rl
WHERE ma.avg_rating = rl.max_rating
   OR ma.avg_rating = rl.min_rating
ORDER BY ma.year, ma.title;"

2. Найти 10 самых свежих оценок от разных пользователей. Вывести название фильма, имя пользователя, оценку и дату отзыва в формате ГГГГ-ММ-ДД.
--------------------------------------------------
WITH latest_ratings AS (
    SELECT
        r.user_id,
        r.movie_id,
        r.rating,
        r.timestamp,
        ROW_NUMBER() OVER (
            PARTITION BY r.user_id
            ORDER BY r.timestamp DESC
        ) AS rn
    FROM ratings r
)
SELECT
    m.title,
    u.name AS user_name,
    lr.rating,
    date(lr.timestamp, 'unixepoch') AS rating_date
FROM latest_ratings lr
JOIN users u ON u.id = lr.user_id
JOIN movies m ON m.id = lr.movie_id
WHERE lr.rn = 1
ORDER BY lr.timestamp DESC
LIMIT 10;

3. Вывести в одном списке все фильмы с максимальным и минимальным средним рейтингом. Отсортировать по году выпуска и названию фильма. В колонке Рекомендуем указать Да для максимального рейтинга и Нет для минимального.
--------------------------------------------------
WITH movie_averages AS (
    SELECT
        m.id,
        m.title,
        m.year,
        AVG(r.rating) AS avg_rating
    FROM movies m
    JOIN ratings r ON r.movie_id = m.id
    GROUP BY m.id, m.title, m.year
),
rating_limits AS (
    SELECT
        MAX(avg_rating) AS max_rating,
        MIN(avg_rating) AS min_rating
    FROM movie_averages
)
SELECT
    ma.title,
    ma.year,
    ROUND(ma.avg_rating, 3) AS avg_rating,
    CASE
        WHEN ma.avg_rating = rl.max_rating THEN 'Да'
        ELSE 'Нет'
    END AS 'Рекомендуем'
FROM movie_averages ma
CROSS JOIN rating_limits rl
WHERE ma.avg_rating = rl.max_rating
   OR ma.avg_rating = rl.min_rating
ORDER BY ma.year, ma.title;

4. Вычислить количество оценок и среднюю оценку, которую дали фильмам пользователи-женщины в период с 2010 по 2012 год.

SELECT
    COUNT(*) AS ratings_count,
    ROUND(AVG(r.rating), 3) AS average_rating
FROM ratings r
JOIN users u ON u.id = r.user_id
WHERE u.gender = 'female'
  AND date(r.timestamp, 'unixepoch') >= '2010-01-01'
  AND date(r.timestamp, 'unixepoch') < '2013-01-01';

5. Составить список фильмов с указанием их средней оценки и места в рейтинге по средней оценке. Отсортировать по году выпуска и названию фильма. Оставить первые 20 записей.
--------------------------------------------------
WITH movie_ratings AS (
    SELECT
        m.id,
        m.title,
        m.year,
        AVG(r.rating) AS avg_rating
    FROM movies m
    JOIN ratings r ON r.movie_id = m.id
    GROUP BY m.id, m.title, m.year
),
ranked_movies AS (
    SELECT
        title,
        year,
        avg_rating,
        RANK() OVER (ORDER BY avg_rating DESC) AS rating_rank
    FROM movie_ratings
)
SELECT
    title,
    year,
    ROUND(avg_rating, 3) AS avg_rating,
    rating_rank
FROM ranked_movies
ORDER BY year, title
LIMIT 20;

7. С помощью рекурсивного CTE составить таблицу умножения для чисел от 1 до 10. Результат должен быть представлен одним столбцом.
--------------------------------------------------
WITH RECURSIVE numbers(n) AS (
    SELECT 1
    UNION ALL
    SELECT n + 1
    FROM numbers
    WHERE n < 100
)
SELECT
    printf(
        '%dx%d=%d',
        (n - 1) / 10 + 1,
        (n - 1) % 10 + 1,
        ((n - 1) / 10 + 1) * ((n - 1) % 10 + 1)
    ) AS multiplication
FROM numbers
ORDER BY n;

WITH RECURSIVE genre_parts(rest, genre) AS (
    SELECT genres || '|', ''
    FROM movies
    WHERE genres IS NOT NULL
      AND genres <> ''
      AND genres <> '(no genres listed)'

    UNION ALL

    SELECT
        substr(rest, instr(rest, '|') + 1),
        substr(rest, 1, instr(rest, '|') - 1)
    FROM genre_parts
    WHERE rest <> ''
)
SELECT DISTINCT genre
FROM genre_parts
WHERE genre <> ''
ORDER BY genre;
